158    def normalize_shape(self, obj, obj_id="unknown"):
159        """Validate and normalise an object's collision shape configuration.
160
161        Supported shape types are BOX, SPHERE, CYLINDER, and CONE. Unknown
162        types and incorrectly sized dimension lists are replaced with safe
163        default BOX or shape-specific values.
164
165        Args:
166            obj (dict): Object configuration containing a ``shape`` entry.
167            obj_id (str): Object identifier used when reporting validation
168            warnings.
169
170        Returns:
171            dict: The object configuration with a normalised ``shape`` entry.
172        """
173        
174        shape = obj.get("shape", {})
175        stype = shape.get("type", "BOX").upper()
176        
177        shape_info = { 
178        "BOX": {"count": 3, "default": [0.05, 0.05, 0.05]},
179        "SPHERE": {"count": 1, "default": [0.05]},
180        "CYLINDER": {"count": 2, "default": [0.05, 0.02]},
181        "CONE": {"count": 2, "default": [0.05, 0.02]}
182        }
183        
184        if stype not in shape_info: #default to BOX if unknown shape type
185            self.get_logger().warn(f"Object '{obj_id}': unknown shape type '{stype}', defaulting to BOX")
186            stype = "BOX"
187
188        info = shape_info[stype]
189        dims = shape.get("dimensions", [])
190        
191        if len(dims) != info["count"]:
192            self.get_logger().warn(
193                f"Object '{obj_id}': expected {info['count']} dimensions for '{stype}', recieved {len(dims)}. "
194                f"Using defaults: {info['default']}"
195            )
196            dims = info["default"].copy()
197        
198        # Store shape
199        obj['shape'] = {
200            'type': stype,
201            'dimensions': dims
202        }
203        return obj

205    def parse_object_data(self, obj_id, obj_data):
206        """Normalise an object's pose and collision shape configuration.
207
208        Orientations supplied as roll, pitch, and yaw are converted to
209        quaternion form. Missing or invalid orientations are replaced with the
210        identity quaternion. Shape information is normalised using
211        :meth:`normalize_shape`.
212
213        Args:
214            obj_id (str): Identifier of the object being parsed.
215            obj_data (dict): Raw object configuration loaded from JSON.
216
217        Returns:
218            dict: The normalised object configuration.
219        """
220        pose = obj_data.get('pose', {})
221        orientation = pose.get('orientation', {})
222        
223        if not orientation:
224            pose['orientation'] = {'x': 0.0, 'y': 0.0, 'z': 0.0, 'w': 1.0}
225        elif any(k in orientation for k in ('roll', 'pitch', 'yaw')):
226            roll = orientation.get('roll', 0.0)
227            pitch = orientation.get('pitch', 0.0)
228            yaw = orientation.get('yaw', 0.0)
229            qx, qy, qz, qw = euler_to_quaternion(roll, pitch, yaw)
230            obj_data['pose']['orientation'] = {'x': qx, 'y': qy, 'z': qz, 'w': qw}
231        else:
232            # Already in quaternion format or invalid
233            if not all(k in orientation for k in ('x', 'y', 'z', 'w')):
234                self.get_logger().warn(f"Object '{obj_id}': invalid orientation format, defaulting to quaternion")
235                pose['orientation'] = {'x': 0.0, 'y': 0.0, 'z': 0.0, 'w': 1.0}
236        
237        # Parse SHAPE (type & dimensions)
238        obj_data = self.normalize_shape(obj_data, obj_id)
239        
240        return obj_data

478    def attach_object_callback(self, request, response):
479        """Attach an environment object to the robot's tool frame.
480
481        Looks up the requested object, obtains its current transform relative
482        to the robot tool, and applies a MoveIt planning-scene update to attach
483        the object to the tool frame.
484
485        Args:
486            request (AttachObject.Request): Service request containing the
487                object identifier.
488            response (AttachObject.Response): Service response populated with
489                the result of the attachment operation.
490
491        Returns:
492            AttachObject.Response: The populated service response indicating
493                whether the object was successfully attached.
494        """
495        
496        obj_id = request.object_id
497        if obj_id in self.attached_objects:
498            self.get_logger().warn(f"Object '{obj_id}' is already attached.")
499            response.success = True
500            response.message = "Object already attached"
501            return response
502
503        if obj_id not in self.static_objects and obj_id not in self.obstacles:
504            response.success = False
505            response.message = f"Object '{obj_id}' not found in environment"
506            self.get_logger().error(response.message)
507            return response
508
509        obj_data = self.static_objects.get(obj_id) or self.obstacles.get(obj_id)
510
511        try:
512            transform = self.tf_buffer.lookup_transform(
513                TOOL_FRAME, BASE_FRAME, rclpy.time.Time(), timeout=Duration(seconds=2.0))
514        except Exception as e:
515            self.get_logger().error(f"Could not look up tool_frame to attach '{obj_id}': {e}")
516            response.success = False
517            response.message = f"Failed to attach '{obj_id}': tool_frame transform unavailable"
518            return response
519
520        relative_pose = pose_in_new_frame(obj_data['pose'], transform)
521        success = self.apply_attach_diff(obj_id, obj_data, relative_pose)
522        if success:
523            self.attached_objects.add(obj_id)
524            response.success = True
525            response.message = f"Object '{obj_id}' attached to gripper"
526        else:
527            response.success = False
528            response.message = f"Failed to attach '{obj_id}'"
529        return response

599    def apply_attach_diff(self, obj_id, obj_data, relative_pose):
600        """Attach a collision object to the robot tool frame in MoveIt.
601
602        Creates an AttachedCollisionObject using the object's collision geometry
603        and pose relative to the tool frame, then submits the planning-scene
604        difference to MoveIt.
605
606        Args:
607            obj_id (str): Identifier of the object to attach.
608            obj_data (dict): Object configuration containing pose and shape data.
609            relative_pose (dict): Object pose expressed relative to the tool
610                frame.
611
612        Returns:
613            bool: True if MoveIt successfully applied the planning-scene change,
614                otherwise False.
615        """
616        
617        if not self.scene_client.wait_for_service(timeout_sec=5.0):
618            self.get_logger().error('/apply_planning_scene not available')
619            return False
620
621        attached_obj_data = {'shape': obj_data['shape'], 'pose': relative_pose}
622        collision_obj = self.build_collision_object(obj_id, attached_obj_data)
623        if collision_obj is None:
624            return False
625        collision_obj.header.frame_id = TOOL_FRAME
626
627        attached = AttachedCollisionObject()
628        attached.link_name = TOOL_FRAME
629        attached.object = collision_obj
630        attached.touch_links = GRIPPER_TOUCH_LINKS
631
632        # MoveIt moves a same-ID object from world to attached automatically
633        # when it sees an AttachedCollisionObject with operation=ADD, no
634        # separate world REMOVE needed, that was fighting this and getting
635        # rejected as "object does not exist" (attach already removed it).
636        scene = PlanningScene()
637        scene.is_diff = True
638        scene.robot_state.is_diff = True
639        scene.robot_state.attached_collision_objects.append(attached)
640
641        return self.send_apply_planning_scene(scene, obj_id, "attach")

674    def send_apply_planning_scene(self, scene, obj_id, label):
675        """Submit a planning-scene update and wait for its result.
676
677        The asynchronous MoveIt service call is given a maximum five-second
678        wait period.
679
680        Args:
681            scene (PlanningScene): Planning-scene diff to apply.
682            obj_id (str): Object identifier used for diagnostic logging.
683            label (str): Operation label used for diagnostic logging.
684
685        Returns:
686            bool: True if the service completed successfully, otherwise False.
687        """
688        
689        request = ApplyPlanningScene.Request()
690        request.scene = scene
691
692        future = self.scene_client.call_async(request)
693        start = time.time()
694        while rclpy.ok() and not future.done():
695            if time.time() - start > 5.0:
696                self.get_logger().error(f"Timed out waiting for apply_planning_scene ({label} {obj_id})")
697                return False
698            time.sleep(0.01)
699
700        if future.result() is not None and future.result().success:
701            return True
702        else:
703            self.get_logger().error(f"apply_planning_scene failed for '{obj_id}' ({label})")
704            return False

745    def build_collision_object(self, obj_id, obj_data):
746        """Construct a MoveIt collision object from environment configuration.
747
748        The configured primitive shape, dimensions, position, and orientation
749        are converted into a ``CollisionObject`` using ``BASE_FRAME`` as the
750        reference frame.
751
752        Args:
753            obj_id (str): Identifier assigned to the collision object.
754            obj_data (dict): Object configuration containing pose and shape data.
755
756        Returns:
757            CollisionObject or None: The constructed collision object, or
758                ``None`` if the configured shape type is unsupported.
759        """   
760        
761        collision_obj = CollisionObject()
762        collision_obj.header.frame_id = BASE_FRAME
763        collision_obj.id = obj_id
764
765        #Shape
766        shape = SolidPrimitive()
767        shape_type_map = {
768            "BOX": SolidPrimitive.BOX,
769            "SPHERE": SolidPrimitive.SPHERE,
770            "CYLINDER": SolidPrimitive.CYLINDER,
771            "CONE": SolidPrimitive.CONE
772        }
773        stype = obj_data['shape']['type']
774        if stype not in shape_type_map:
775            self.get_logger().error(f"Object '{obj_id}': unknown shape type '{stype}'")
776            return None
777        shape.type = shape_type_map[stype]
778        shape.dimensions = obj_data['shape']['dimensions']
779
780        # Pose
781        pose = Pose()
782        # Position
783        pos = obj_data['pose']['position']
784        pose.position.x = pos['x']
785        pose.position.y = pos['y']
786        pose.position.z = pos['z']
787        # Orientation
788        orient = obj_data['pose']['orientation']
789        pose.orientation.x = orient['x']
790        pose.orientation.y = orient['y']
791        pose.orientation.z = orient['z']
792        pose.orientation.w = orient['w']
793
794        # Build collision object
795        collision_obj.primitives.append(shape)
796        collision_obj.primitive_poses.append(pose)
797        collision_obj.operation = CollisionObject.ADD
798
799        return collision_obj

801    def publish_planning_scene(self):
802        """Wait for MoveIt and publish the initial planning scene.
803
804        This method is executed by the node's background scene thread. It waits
805        for the ``/apply_planning_scene`` service to become available before
806        calling :meth:`apply_full_scene`.
807        """
808        
809        self.get_logger().info('Waiting for /apply_planning_scene service...')
810        if not self.scene_client.wait_for_service(timeout_sec=15.0):
811            self.get_logger().warn('/apply_planning_scene not available. MoveIt may not be running. Skipping planning scene setup.')
812            return
813
814        self.get_logger().info('MoveIt ready. Publishing collision objects to planning scene...')
815        time.sleep(1.0)
816        self.apply_full_scene()

818    def apply_full_scene(self):
819        """Rebuild and apply the complete MoveIt planning scene.
820
821        The scene is reconstructed from the current obstacle and static-object
822        dictionaries. Any tracked attached objects are first removed from the
823        attached-object state. On successful application, attachment tracking
824        is cleared.
825
826        Returns:
827            bool: True if the complete planning scene was successfully applied,
828                otherwise False.
829        """
830        
831        scene = PlanningScene()
832        scene.is_diff = True
833
834        if self.attached_objects:
835            scene.robot_state.is_diff = True
836            for obj_id in self.attached_objects:
837                detach_marker = AttachedCollisionObject()
838                detach_marker.link_name = TOOL_FRAME
839                detach_marker.object.id = obj_id
840                detach_marker.object.operation = CollisionObject.REMOVE
841                scene.robot_state.attached_collision_objects.append(detach_marker)
842
843        # Add obstacles from obstacles dictionary
844        for obs_id, obs_data in self.obstacles.items():
845            obj = self.build_collision_object(obs_id, obs_data)
846            if obj is not None:
847                scene.world.collision_objects.append(obj)
848                self.get_logger().info(f"Adding obstacle: '{obs_id}'")
849
850        # Add objects from object dictionary
851        for obj_id, obj_data in self.static_objects.items():
852            obj = self.build_collision_object(obj_id, obj_data)
853            if obj is not None:
854                scene.world.collision_objects.append(obj)
855                self.get_logger().info(f"Adding object: '{obj_id}'")
856
857        request = ApplyPlanningScene.Request()
858        request.scene = scene
859
860        future = self.scene_client.call_async(request)
861        start = time.time()
862        while rclpy.ok() and not future.done():
863            if time.time() - start > 5.0:
864                self.get_logger().error('Timed out waiting for apply_planning_scene (full scene)')
865                return False
866            time.sleep(0.1)
867
868        if future.result() is not None and future.result().success:
869            self.attached_objects.clear()
870            total = len(self.obstacles) + len(self.static_objects)
871            self.get_logger().info(f'Planning scene updated with {total} collision objects.')
872            for obs_id, obs_data in self.obstacles.items():
873                pos = obs_data['pose']['position']
874                self.get_logger().info(f"  Obstacle '{obs_id}' at x={pos['x']}, y={pos['y']}, z={pos['z']}")
875            for obj_id, obj_data in self.static_objects.items():
876                pos = obj_data['pose']['position']
877                self.get_logger().info(f"  Object '{obj_id}' at x={pos['x']}, y={pos['y']}, z={pos['z']}")
878            return True
879        else:
880            self.get_logger().error('Failed to apply planning scene.')
881            return False

159    def normalize_shape(self, obj, obj_id="unknown"):
160        """Validate and normalise an object's collision shape configuration.
161
162        Supported shape types are BOX, SPHERE, CYLINDER, and CONE. Unknown
163        types and incorrectly sized dimension lists are replaced with safe
164        default BOX or shape-specific values.
165
166        Args:
167            obj (dict): Object configuration containing a ``shape`` entry.
168            obj_id (str): Object identifier used when reporting validation
169            warnings.
170
171        Returns:
172            dict: The object configuration with a normalised ``shape`` entry.
173        """
174        
175        shape = obj.get("shape", {})
176        stype = shape.get("type", "BOX").upper()
177        
178        shape_info = { 
179        "BOX": {"count": 3, "default": [0.05, 0.05, 0.05]},
180        "SPHERE": {"count": 1, "default": [0.05]},
181        "CYLINDER": {"count": 2, "default": [0.05, 0.02]},
182        "CONE": {"count": 2, "default": [0.05, 0.02]}
183        }
184        
185        if stype not in shape_info: #default to BOX if unknown shape type
186            self.get_logger().warn(f"Object '{obj_id}': unknown shape type '{stype}', defaulting to BOX")
187            stype = "BOX"
188
189        info = shape_info[stype]
190        dims = shape.get("dimensions", [])
191        
192        if len(dims) != info["count"]:
193            self.get_logger().warn(
194                f"Object '{obj_id}': expected {info['count']} dimensions for '{stype}', recieved {len(dims)}. "
195                f"Using defaults: {info['default']}"
196            )
197            dims = info["default"].copy()
198        
199        # Store shape
200        obj['shape'] = {
201            'type': stype,
202            'dimensions': dims
203        }
204        return obj

206    def parse_object_data(self, obj_id, obj_data):
207        """Normalise an object's pose and collision shape configuration.
208
209        Orientations supplied as roll, pitch, and yaw are converted to
210        quaternion form. Missing or invalid orientations are replaced with the
211        identity quaternion. Shape information is normalised using
212        :meth:`normalize_shape`.
213
214        Args:
215            obj_id (str): Identifier of the object being parsed.
216            obj_data (dict): Raw object configuration loaded from JSON.
217
218        Returns:
219            dict: The normalised object configuration.
220        """
221        pose = obj_data.get('pose', {})
222        orientation = pose.get('orientation', {})
223        
224        if not orientation:
225            pose['orientation'] = {'x': 0.0, 'y': 0.0, 'z': 0.0, 'w': 1.0}
226        elif any(k in orientation for k in ('roll', 'pitch', 'yaw')):
227            roll = orientation.get('roll', 0.0)
228            pitch = orientation.get('pitch', 0.0)
229            yaw = orientation.get('yaw', 0.0)
230            qx, qy, qz, qw = euler_to_quaternion(roll, pitch, yaw)
231            obj_data['pose']['orientation'] = {'x': qx, 'y': qy, 'z': qz, 'w': qw}
232        else:
233            # Already in quaternion format or invalid
234            if not all(k in orientation for k in ('x', 'y', 'z', 'w')):
235                self.get_logger().warn(f"Object '{obj_id}': invalid orientation format, defaulting to quaternion")
236                pose['orientation'] = {'x': 0.0, 'y': 0.0, 'z': 0.0, 'w': 1.0}
237        
238        # Parse SHAPE (type & dimensions)
239        obj_data = self.normalize_shape(obj_data, obj_id)
240        
241        return obj_data

600    def apply_attach_diff(self, obj_id, obj_data, relative_pose):
601        """Attach a collision object to the robot tool frame in MoveIt.
602
603        Creates an AttachedCollisionObject using the object's collision geometry
604        and pose relative to the tool frame, then submits the planning-scene
605        difference to MoveIt.
606
607        Args:
608            obj_id (str): Identifier of the object to attach.
609            obj_data (dict): Object configuration containing pose and shape data.
610            relative_pose (dict): Object pose expressed relative to the tool
611                frame.
612
613        Returns:
614            bool: True if MoveIt successfully applied the planning-scene change,
615                otherwise False.
616        """
617        
618        if not self.scene_client.wait_for_service(timeout_sec=5.0):
619            self.get_logger().error('/apply_planning_scene not available')
620            return False
621
622        attached_obj_data = {'shape': obj_data['shape'], 'pose': relative_pose}
623        collision_obj = self.build_collision_object(obj_id, attached_obj_data)
624        if collision_obj is None:
625            return False
626        collision_obj.header.frame_id = TOOL_FRAME
627
628        attached = AttachedCollisionObject()
629        attached.link_name = TOOL_FRAME
630        attached.object = collision_obj
631        attached.touch_links = GRIPPER_TOUCH_LINKS
632
633        # MoveIt moves a same-ID object from world to attached automatically
634        # when it sees an AttachedCollisionObject with operation=ADD, no
635        # separate world REMOVE needed, that was fighting this and getting
636        # rejected as "object does not exist" (attach already removed it).
637        scene = PlanningScene()
638        scene.is_diff = True
639        scene.robot_state.is_diff = True
640        scene.robot_state.attached_collision_objects.append(attached)
641
642        return self.send_apply_planning_scene(scene, obj_id, "attach")

746    def build_collision_object(self, obj_id, obj_data):
747        """Construct a MoveIt collision object from environment configuration.
748
749        The configured primitive shape, dimensions, position, and orientation
750        are converted into a ``CollisionObject`` using ``BASE_FRAME`` as the
751        reference frame.
752
753        Args:
754            obj_id (str): Identifier assigned to the collision object.
755            obj_data (dict): Object configuration containing pose and shape data.
756
757        Returns:
758            CollisionObject or None: The constructed collision object, or
759                ``None`` if the configured shape type is unsupported.
760        """   
761        
762        collision_obj = CollisionObject()
763        collision_obj.header.frame_id = BASE_FRAME
764        collision_obj.id = obj_id
765
766        #Shape
767        shape = SolidPrimitive()
768        shape_type_map = {
769            "BOX": SolidPrimitive.BOX,
770            "SPHERE": SolidPrimitive.SPHERE,
771            "CYLINDER": SolidPrimitive.CYLINDER,
772            "CONE": SolidPrimitive.CONE
773        }
774        stype = obj_data['shape']['type']
775        if stype not in shape_type_map:
776            self.get_logger().error(f"Object '{obj_id}': unknown shape type '{stype}'")
777            return None
778        shape.type = shape_type_map[stype]
779        shape.dimensions = obj_data['shape']['dimensions']
780
781        # Pose
782        pose = Pose()
783        # Position
784        pos = obj_data['pose']['position']
785        pose.position.x = pos['x']
786        pose.position.y = pos['y']
787        pose.position.z = pos['z']
788        # Orientation
789        orient = obj_data['pose']['orientation']
790        pose.orientation.x = orient['x']
791        pose.orientation.y = orient['y']
792        pose.orientation.z = orient['z']
793        pose.orientation.w = orient['w']
794
795        # Build collision object
796        collision_obj.primitives.append(shape)
797        collision_obj.primitive_poses.append(pose)
798        collision_obj.operation = CollisionObject.ADD
799
800        return collision_obj

819    def apply_full_scene(self):
820        """Rebuild and apply the complete MoveIt planning scene.
821
822        The scene is reconstructed from the current obstacle and static-object
823        dictionaries. Any tracked attached objects are first removed from the
824        attached-object state. On successful application, attachment tracking
825        is cleared.
826
827        Returns:
828            bool: True if the complete planning scene was successfully applied,
829                otherwise False.
830        """
831        
832        scene = PlanningScene()
833        scene.is_diff = True
834
835        if self.attached_objects:
836            scene.robot_state.is_diff = True
837            for obj_id in self.attached_objects:
838                detach_marker = AttachedCollisionObject()
839                detach_marker.link_name = TOOL_FRAME
840                detach_marker.object.id = obj_id
841                detach_marker.object.operation = CollisionObject.REMOVE
842                scene.robot_state.attached_collision_objects.append(detach_marker)
843
844        # Add obstacles from obstacles dictionary
845        for obs_id, obs_data in self.obstacles.items():
846            obj = self.build_collision_object(obs_id, obs_data)
847            if obj is not None:
848                scene.world.collision_objects.append(obj)
849                self.get_logger().info(f"Adding obstacle: '{obs_id}'")
850
851        # Add objects from object dictionary
852        for obj_id, obj_data in self.static_objects.items():
853            obj = self.build_collision_object(obj_id, obj_data)
854            if obj is not None:
855                scene.world.collision_objects.append(obj)
856                self.get_logger().info(f"Adding object: '{obj_id}'")
857
858        request = ApplyPlanningScene.Request()
859        request.scene = scene
860
861        future = self.scene_client.call_async(request)
862        start = time.time()
863        while rclpy.ok() and not future.done():
864            if time.time() - start > 5.0:
865                self.get_logger().error('Timed out waiting for apply_planning_scene (full scene)')
866                return False
867            time.sleep(0.1)
868
869        if future.result() is not None and future.result().success:
870            self.attached_objects.clear()
871            total = len(self.obstacles) + len(self.static_objects)
872            self.get_logger().info(f'Planning scene updated with {total} collision objects.')
873            for obs_id, obs_data in self.obstacles.items():
874                pos = obs_data['pose']['position']
875                self.get_logger().info(f"  Obstacle '{obs_id}' at x={pos['x']}, y={pos['y']}, z={pos['z']}")
876            for obj_id, obj_data in self.static_objects.items():
877                pos = obj_data['pose']['position']
878                self.get_logger().info(f"  Object '{obj_id}' at x={pos['x']}, y={pos['y']}, z={pos['z']}")
879            return True
880        else:
881            self.get_logger().error('Failed to apply planning scene.')
882            return False

749        The configured primitive shape, dimensions, position, and orientation
750        are converted into a ``CollisionObject`` using ``BASE_FRAME`` as the
751        reference frame.
752
753        Args:
754            obj_id (str): Identifier assigned to the collision object.
755            obj_data (dict): Object configuration containing pose and shape data.
756
757        Returns:
758            CollisionObject or None: The constructed collision object, or
759                ``None`` if the configured shape type is unsupported.
760        """   
761        
762        collision_obj = CollisionObject()
763        collision_obj.header.frame_id = BASE_FRAME
764        collision_obj.id = obj_id
765
766        #Shape
767        shape = SolidPrimitive()
768        shape_type_map = {
769            "BOX": SolidPrimitive.BOX,
770            "SPHERE": SolidPrimitive.SPHERE,
771            "CYLINDER": SolidPrimitive.CYLINDER,
772            "CONE": SolidPrimitive.CONE
773        }
774        stype = obj_data['shape']['type']
775        if stype not in shape_type_map:
776            self.get_logger().error(f"Object '{obj_id}': unknown shape type '{stype}'")
777            return None
778        shape.type = shape_type_map[stype]
779        shape.dimensions = obj_data['shape']['dimensions']
780
781        # Pose
782        pose = Pose()
783        # Position
784        pos = obj_data['pose']['position']
785        pose.position.x = pos['x']
786        pose.position.y = pos['y']
787        pose.position.z = pos['z']

791        pose.orientation.y = orient['y']
792        pose.orientation.z = orient['z']
793        pose.orientation.w = orient['w']
