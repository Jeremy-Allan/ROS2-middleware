 208    def list_controllers_callback(self, future):
 209        """Process the controller-manager health check response.
 210
 211        Checks whether ``fault_controller`` exists and whether it is active.
 212        Updates the node status and warning state when the controller becomes
 213        inactive or returns to an active state.
 214
 215        Args:
 216            future: Future containing the ``ListControllers`` service response.
 217
 218        Returns:
 219            None
 220        """
 221        
 222        try:
 223            response = future.result()
 224            fault_ctrl_active = False
 225            fault_ctrl_found = False
 226            for controller in response.controller:
 227                if controller.name == 'fault_controller':
 228                    fault_ctrl_found = True
 229                    if controller.state == 'active':
 230                        fault_ctrl_active = True
 231                    break
 232
 233            if fault_ctrl_found and not fault_ctrl_active:
 234                if not self.fault_controller_warning_active:
 235                    self.fault_controller_warning_active = True
 236                    self.get_logger().warn("fault_controller is present but NOT active!")
 237                    self.status_text = "SYSTEM CONFIG WARNING: fault_controller is NOT active"
 238                    self.publish_status()
 239            elif fault_ctrl_found and fault_ctrl_active:
 240                if self.fault_controller_warning_active:
 241                    self.fault_controller_warning_active = False
 242                    self.get_logger().info("fault_controller is now active! Clearing config warning status.")
 243                    if self.status_text == "SYSTEM CONFIG WARNING: fault_controller is NOT active":
 244                        self.status_text = "Hardware Interface Client Ready"
 245                    self.publish_status()
 246        except Exception as e:
 247            self.get_logger().error(f"Failed to query controllers health: {e}")

 318    def _await_action(self, event: threading.Event, timeout_sec: float, success_attr: str, message_attr: str, action_desc: str, response):
 319        """Wait for an action to finish and populate a service response.
 320
 321        Waits for the supplied event for up to ``timeout_sec`` seconds and reads
 322        the configured success and message state when the action completes.
 323
 324        Args:
 325            event (threading.Event): Event that is set when the action completes.
 326            timeout_sec (float): Maximum time to wait for the action.
 327            success_attr (str): Attribute name containing the action success state,
 328                or a callable returning that state.
 329            message_attr (str): Attribute name containing the action result message,
 330                or a callable returning that message.
 331            action_desc (str): Human-readable description used in timeout messages.
 332            response: ROS 2 service response object to populate.
 333
 334        Returns:
 335            The supplied service response with its ``success`` and ``message``
 336            fields populated.
 337        """
 338
 339        # TODO: create a request/response interface to do type constraints in functions
 340        finished = event.wait(timeout=timeout_sec)
 341        if not finished:
 342            response.success = False
 343            response.message = f"{action_desc} timed out after {timeout_sec}s"
 344            self.get_logger().error(response.message)
 345        else:
 346            response.success = getattr(self, success_attr) if isinstance(success_attr, str) else success_attr()
 347            response.message = getattr(self, message_attr) if isinstance(message_attr, str) else message_attr()
 348        return response

 387    def handle_joint_move(self, request, response):
 388        """Handle an absolute or relative joint-space movement request.
 389
 390        Absolute mode sends the requested six joint positions directly to MoveIt.
 391        Relative mode adds the requested joint deltas to the latest joint state.
 392
 393        When ``wait_for_completion`` is false, the service waits only for the
 394        short fire-and-forget rejection window before returning, allowing the
 395        caller to continue while the movement remains in progress.
 396
 397        Args:
 398            request (JointMove.Request): Service request containing joint target
 399                positions, relative-mode settings, completion behaviour, and
 400                movement parameters.
 401            response (JointMove.Response): Service response populated with the
 402                movement result.
 403
 404        Returns:
 405            JointMove.Response: The populated service response.
 406        """
 407
 408        joint_positions = list(request.joint_positions)
 409
 410        if request.relative:
 411            if self.latest_joint_positions is None:
 412                response.success = False
 413                response.message = "No joint state available for relative joint move"
 414                return self.finalize_service_status(response)
 415            try:
 416                current = [self.latest_joint_positions[name] for name in JOINT_NAMES]
 417            except KeyError as e:
 418                response.success = False
 419                response.message = f"Missing joint {e} in latest joint state"
 420                return self.finalize_service_status(response)
 421            joint_positions = [current[i] + joint_positions[i] for i in range(6)]
 422
 423        self.get_logger().info(f"Service Call: Joint Move to {joint_positions}")
 424        self.current_state = ExtendedStatus.STATE_BUSY
 425        self.status_text = f"Moving to joint targets {joint_positions}..."
 426        self.publish_status()
 427
 428        if not self.send_joint_goal(joint_positions, motion_params=request.motion_params):
 429            response.success = False
 430            response.message = "Failed to initiate joint move"
 431            return self.finalize_service_status(response)
 432
 433        if not request.wait_for_completion:
 434            if self.arm_movement_finished.wait(timeout=self.FIRE_AND_FORGET_REJECTION_WINDOW_SEC):
 435                response.success = self.arm_action_successful
 436                response.message = self.arm_action_message
 437                return self.finalize_service_status(response)
 438            response.success = True
 439            response.message = "Joint move goal accepted (not waiting for completion)"
 440            return response
 441
 442        self._await_action(
 443            self.arm_movement_finished,
 444            self.ACTION_TIMEOUT_SEC,
 445            'arm_action_successful',
 446            'arm_action_message',
 447            f"Joint move to {joint_positions}",
 448            response
 449        )
 450        return self.finalize_service_status(response)

 498    def handle_relative_move(self, request, response):
 499        """Handle a relative Cartesian movement request.
 500
 501        Looks up the current tool pose using TF, applies the requested position
 502        and optional roll, pitch, and yaw offsets, and sends the resulting target
 503        pose to MoveIt.
 504
 505        Args:
 506            request (RelativeMove.Request): Service request containing position
 507                and optional orientation deltas and movement parameters.
 508            response (RelativeMove.Response): Service response populated with the
 509                movement result.
 510
 511        Returns:
 512            RelativeMove.Response: The populated service response.
 513        """
 514        
 515        vx = request.vx
 516        vy = request.vy
 517        vz = request.vz
 518        self.get_logger().info(f"Service Call: Relative Move by Vector [{vx}, {vy}, {vz}]")
 519        self.current_state = ExtendedStatus.STATE_BUSY
 520        self.status_text = f"Executing relative move by vector [{vx}, {vy}, {vz}]..."
 521        self.publish_status()
 522
 523        try:
 524            # Look up current pose of the tool frame
 525            now = rclpy.time.Time()
 526            trans = self.tf_buffer.lookup_transform(BASE_FRAME, TOOL_FRAME, now, timeout=rclpy.duration.Duration(seconds=1.0))
 527
 528            curr_x = trans.transform.translation.x
 529            curr_y = trans.transform.translation.y
 530            curr_z = trans.transform.translation.z
 531
 532            target_x = curr_x + vx
 533            target_y = curr_y + vy
 534            target_z = curr_z + vz
 535
 536            self.get_logger().info(f"Calculated target: {target_x:.3f}, {target_y:.3f}, {target_z:.3f}")
 537
 538            has_orientation = request.has_orientation
 539            target_roll = target_pitch = target_yaw = 0.0
 540            if has_orientation:
 541                # quaternion_to_euler uses asin() for pitch, capped at +-90deg.
 542                # If the arm is already near that boundary (e.g. right after
 543                # tilted_for_pour), roll/yaw become coupled and this delta
 544                # composition gives unintuitive results. Fine for small nudges
 545                # from a normal pose, which is the expected use case.
 546                q = trans.transform.rotation
 547                curr_roll, curr_pitch, curr_yaw = quaternion_to_euler(q.x, q.y, q.z, q.w)
 548                target_roll = curr_roll + request.roll_delta
 549                target_pitch = curr_pitch + request.pitch_delta
 550                target_yaw = curr_yaw + request.yaw_delta
 551                self.get_logger().info(
 552                    f"Calculated target orientation (rpy): {target_roll}, {target_pitch}, {target_yaw}"
 553                )
 554
 555            if self.send_goal(
 556                target_x, target_y, target_z,
 557                has_orientation=has_orientation,
 558                roll=target_roll,
 559                pitch=target_pitch,
 560                yaw=target_yaw,
 561                motion_params=request.motion_params,
 562            ):
 563                self._await_action(
 564                    self.arm_movement_finished,
 565                    self.ACTION_TIMEOUT_SEC,
 566                    'arm_action_successful',
 567                    'arm_action_message',
 568                    f"Relative movement to ({target_x}, {target_y}, {target_z})",
 569                    response
 570                )
 571            else:
 572                response.success = False
 573                response.message = "Failed to initiate relative movement (action server unavailable)"
 574
 575        except Exception as e:
 576            self.get_logger().error(f"Could not calculate relative move: {e}")
 577            response.success = False
 578            response.message = f"Relative move TF lookup failed: {e}"
 579
 580        return self.finalize_service_status(response)

 620    def clamp_motion_params(self, motion_params):
 621        """Clamp velocity and acceleration scaling factors to the valid range.
 622
 623        Values outside the MoveIt scaling range of 0.0 to 1.0 are clamped and
 624        a warning is logged.
 625
 626        Args:
 627            motion_params: Movement parameter object containing
 628                ``velocity_scale`` and ``acceleration_scale``.
 629
 630        Returns:
 631            tuple[float, float]: Clamped velocity and acceleration scaling
 632                factors.
 633        """
 634
 635        velocity_scale = motion_params.velocity_scale
 636        acceleration_scale = motion_params.acceleration_scale
 637
 638        if velocity_scale < 0.0 or velocity_scale > 1.0:
 639            self.get_logger().warn(f"velocity_scale {velocity_scale} out of range, clamping to [0.0, 1.0]")
 640            velocity_scale = max(0.0, min(1.0, velocity_scale))
 641
 642        if acceleration_scale < 0.0 or acceleration_scale > 1.0:
 643            self.get_logger().warn(f"acceleration_scale {acceleration_scale} out of range, clamping to [0.0, 1.0]")
 644            acceleration_scale = max(0.0, min(1.0, acceleration_scale))
 645
 646        return velocity_scale, acceleration_scale

 649    def send_goal(self, x, y, z, has_orientation=False, roll=0.0, pitch=0.0, yaw=0.0, motion_params=None):
 650        """Send a Cartesian movement goal to MoveIt.
 651
 652        Constructs a position constraint for the robot tool and optionally adds
 653        an orientation constraint and motion-scaling parameters. The goal is
 654        submitted asynchronously.
 655
 656        This method reports whether the goal could be initiated; it does not
 657        indicate whether the physical movement has completed successfully.
 658        Completion is reported through the action callbacks.
 659
 660        Args:
 661            x (float): Target X position in ``BASE_FRAME``.
 662            y (float): Target Y position in ``BASE_FRAME``.
 663            z (float): Target Z position in ``BASE_FRAME``.
 664            has_orientation (bool): Whether an orientation constraint should be
 665                included.
 666            roll (float): Target roll angle in radians.
 667            pitch (float): Target pitch angle in radians.
 668            yaw (float): Target yaw angle in radians.
 669            motion_params: Optional movement parameters containing velocity and
 670                acceleration scaling factors.
 671
 672        Returns:
 673            bool: ``True`` if the MoveIt action server is available and the goal
 674                was submitted; ``False`` if the action server is unavailable.
 675        """
 676
 677        if not self.arm_client.wait_for_server(timeout_sec=self.SERVER_WAIT_TIMEOUT_SEC):
 678            self.get_logger().error('Arm server not available')
 679            return False
 680
 681        goal_msg = MoveGroup.Goal()
 682        goal_msg.request.group_name = self.PLANNING_GROUP
 683        goal_msg.request.num_planning_attempts = self.NUM_PLANNING_ATTEMPTS
 684        goal_msg.request.allowed_planning_time = self.ALLOWED_PLANNING_TIME_SEC
 685
 686        pos_constraint = PositionConstraint()
 687        pos_constraint.header.frame_id = BASE_FRAME
 688        pos_constraint.link_name = TOOL_FRAME
 689
 690        sphere = SolidPrimitive()
 691        sphere.type = SolidPrimitive.SPHERE
 692        sphere.dimensions = [self.SPHERE_TOLERANCE_RADIUS]
 693
 694        target_pose = Pose()
 695        target_pose.position.x = float(x)
 696        target_pose.position.y = float(y)
 697        target_pose.position.z = float(z)
 698
 699        pos_constraint.constraint_region.primitives.append(sphere)
 700        pos_constraint.constraint_region.primitive_poses.append(target_pose)
 701        pos_constraint.weight = 1.0
 702
 703        goal_constraints = Constraints()
 704        goal_constraints.position_constraints.append(pos_constraint)
 705
 706        if has_orientation:
 707            qx, qy, qz, qw = euler_to_quaternion(roll, pitch, yaw)
 708            orient_constraint = OrientationConstraint()
 709            orient_constraint.header.frame_id = BASE_FRAME
 710            orient_constraint.link_name = TOOL_FRAME
 711            orient_constraint.orientation.x = qx
 712            orient_constraint.orientation.y = qy
 713            orient_constraint.orientation.z = qz
 714            orient_constraint.orientation.w = qw
 715            orient_constraint.absolute_x_axis_tolerance = 0.1
 716            orient_constraint.absolute_y_axis_tolerance = 0.1
 717            orient_constraint.absolute_z_axis_tolerance = 0.1
 718            orient_constraint.weight = 1.0
 719            goal_constraints.orientation_constraints.append(orient_constraint)
 720
 721        goal_msg.request.goal_constraints.append(goal_constraints)
 722
 723        if motion_params is not None:
 724            velocity_scale, acceleration_scale = self.clamp_motion_params(motion_params)
 725            if velocity_scale > 0.0:
 726                goal_msg.request.max_velocity_scaling_factor = velocity_scale
 727            if acceleration_scale > 0.0:
 728                goal_msg.request.max_acceleration_scaling_factor = acceleration_scale
 729
 730        self.arm_movement_finished.clear()
 731        future = self.arm_client.send_goal_async(
 732            goal_msg,
 733            feedback_callback=self.arm_feedback_callback
 734        )
 735        future.add_done_callback(self.goal_response_callback)
 736        return True

 838        """Process the response received after submitting an arm action goal.
 839

 841        result. If the goal is rejected or an exception occurs, records the

 885    def result_callback(self, future):
 886        """Process the final result of a MoveIt arm action.
 887
 888        Maps MoveIt error codes to human-readable status messages, updates the
 889        stored movement result, handles execution failures, and releases the
 890        movement wait event.
 891
 892        Args:
 893            future: Future containing the completed MoveIt action result.
 894
 895        Returns:
 896            None
 897        """
 898
 899        try:
 900            result = future.result().result
 901            error_code = result.error_code.val
 902
 903            if error_code == result.error_code.SUCCESS:
 904                self.get_logger().info('Movement complete!')
 905                self.arm_action_successful = True
 906                self.arm_action_message = 'Movement complete'
 907            else:
 908                self.arm_action_successful = False
 909
 910                match error_code:
 911                    case result.error_code.NO_IK_SOLUTION:
 912                        msg = "Coordinates out of reach (no inverse kinematics solution)"
 913                    case result.error_code.PLANNING_FAILED:
 914                        msg = "Planning failed (path blocked by obstacle or self-collision)"
 915                    case result.error_code.TIMED_OUT:
 916                        msg = "MoveIt planning/movement timed out"
 917                    case result.error_code.GOAL_IN_COLLISION:
 918                        msg = "Goal is in collision (target position inside an obstacle)"
 919                    case result.error_code.START_STATE_IN_COLLISION:
 920                        msg = "Start state is in collision (robot currently in collision)"
 921                    case result.error_code.CONTROL_FAILED:
 922                        msg = "Control failed during execution (hardware error)"
 923                    case result.error_code.ABORT:
 924                        msg = "Movement was aborted by MoveIt"
 925                    case _:
 926                        msg = f"MoveIt failed with error code: {error_code}"
 927
 928                self.get_logger().error(f"ERROR: {msg}")
 929                self.arm_action_message = msg
 930                self.handle_moveit_failure()
 931        except Exception as e:
 932            self.get_logger().error(f"Error processing arm action result: {e}")
 933            self.arm_action_successful = False
 934            self.arm_action_message = f"Action result processing error: {e}"
 935        finally:
 936            # Reset state here (not just in finalize_service_status, which only
 937            # runs for a caller that waited) so a fire-and-forget joint move
 938            # (e.g. 'throw's fling) doesn't leave current_state stuck on BUSY
 939            # once it actually finishes with nobody waiting on it.
 940            if not self.is_faulted:
 941                self.current_state = ExtendedStatus.STATE_IDLE
 942                self.status_text = "Movement complete!" if self.arm_action_successful else "Movement failed"
 943                self.publish_status()
 944            self.arm_movement_finished.set()

 209    def list_controllers_callback(self, future):
 210        """Process the controller-manager health check response.
 211
 212        Checks whether ``fault_controller`` exists and whether it is active.
 213        Updates the node status and warning state when the controller becomes
 214        inactive or returns to an active state.
 215
 216        Args:
 217            future: Future containing the ``ListControllers`` service response.
 218
 219        Returns:
 220            None
 221        """
 222        
 223        try:
 224            response = future.result()
 225            fault_ctrl_active = False
 226            fault_ctrl_found = False
 227            for controller in response.controller:
 228                if controller.name == 'fault_controller':
 229                    fault_ctrl_found = True
 230                    if controller.state == 'active':
 231                        fault_ctrl_active = True
 232                    break
 233
 234            if fault_ctrl_found and not fault_ctrl_active:
 235                if not self.fault_controller_warning_active:
 236                    self.fault_controller_warning_active = True
 237                    self.get_logger().warn("fault_controller is present but NOT active!")
 238                    self.status_text = "SYSTEM CONFIG WARNING: fault_controller is NOT active"
 239                    self.publish_status()
 240            elif fault_ctrl_found and fault_ctrl_active:
 241                if self.fault_controller_warning_active:
 242                    self.fault_controller_warning_active = False
 243                    self.get_logger().info("fault_controller is now active! Clearing config warning status.")
 244                    if self.status_text == "SYSTEM CONFIG WARNING: fault_controller is NOT active":
 245                        self.status_text = "Hardware Interface Client Ready"
 246                    self.publish_status()
 247        except Exception as e:
 248            self.get_logger().error(f"Failed to query controllers health: {e}")

 319    def _await_action(self, event: threading.Event, timeout_sec: float, success_attr: str, message_attr: str, action_desc: str, response):
 320        """Wait for an action to finish and populate a service response.
 321
 322        Waits for the supplied event for up to ``timeout_sec`` seconds and reads
 323        the configured success and message state when the action completes.
 324
 325        Args:
 326            event (threading.Event): Event that is set when the action completes.
 327            timeout_sec (float): Maximum time to wait for the action.
 328            success_attr (str): Attribute name containing the action success state,
 329                or a callable returning that state.
 330            message_attr (str): Attribute name containing the action result message,
 331                or a callable returning that message.
 332            action_desc (str): Human-readable description used in timeout messages.
 333            response: ROS 2 service response object to populate.
 334
 335        Returns:
 336            The supplied service response with its ``success`` and ``message``
 337            fields populated.
 338        """
 339
 340        # TODO: create a request/response interface to do type constraints in functions
 341        finished = event.wait(timeout=timeout_sec)
 342        if not finished:
 343            response.success = False
 344            response.message = f"{action_desc} timed out after {timeout_sec}s"
 345            self.get_logger().error(response.message)
 346        else:
 347            response.success = getattr(self, success_attr) if isinstance(success_attr, str) else success_attr()
 348            response.message = getattr(self, message_attr) if isinstance(message_attr, str) else message_attr()
 349        return response

 388    def handle_joint_move(self, request, response):
 389        """Handle an absolute or relative joint-space movement request.
 390
 391        Absolute mode sends the requested six joint positions directly to MoveIt.
 392        Relative mode adds the requested joint deltas to the latest joint state.
 393
 394        When ``wait_for_completion`` is false, the service waits only for the
 395        short fire-and-forget rejection window before returning, allowing the
 396        caller to continue while the movement remains in progress.
 397
 398        Args:
 399            request (JointMove.Request): Service request containing joint target
 400                positions, relative-mode settings, completion behaviour, and
 401                movement parameters.
 402            response (JointMove.Response): Service response populated with the
 403                movement result.
 404
 405        Returns:
 406            JointMove.Response: The populated service response.
 407        """
 408
 409        joint_positions = list(request.joint_positions)
 410
 411        if request.relative:
 412            if self.latest_joint_positions is None:
 413                response.success = False
 414                response.message = "No joint state available for relative joint move"
 415                return self.finalize_service_status(response)
 416            try:
 417                current = [self.latest_joint_positions[name] for name in JOINT_NAMES]
 418            except KeyError as e:
 419                response.success = False
 420                response.message = f"Missing joint {e} in latest joint state"
 421                return self.finalize_service_status(response)
 422            joint_positions = [current[i] + joint_positions[i] for i in range(6)]
 423
 424        self.get_logger().info(f"Service Call: Joint Move to {joint_positions}")
 425        self.current_state = ExtendedStatus.STATE_BUSY
 426        self.status_text = f"Moving to joint targets {joint_positions}..."
 427        self.publish_status()
 428
 429        if not self.send_joint_goal(joint_positions, motion_params=request.motion_params):
 430            response.success = False
 431            response.message = "Failed to initiate joint move"
 432            return self.finalize_service_status(response)
 433
 434        if not request.wait_for_completion:
 435            if self.arm_movement_finished.wait(timeout=self.FIRE_AND_FORGET_REJECTION_WINDOW_SEC):
 436                response.success = self.arm_action_successful
 437                response.message = self.arm_action_message
 438                return self.finalize_service_status(response)
 439            response.success = True
 440            response.message = "Joint move goal accepted (not waiting for completion)"
 441            return response
 442
 443        self._await_action(
 444            self.arm_movement_finished,
 445            self.ACTION_TIMEOUT_SEC,
 446            'arm_action_successful',
 447            'arm_action_message',
 448            f"Joint move to {joint_positions}",
 449            response
 450        )
 451        return self.finalize_service_status(response)

 499    def handle_relative_move(self, request, response):
 500        """Handle a relative Cartesian movement request.
 501
 502        Looks up the current tool pose using TF, applies the requested position
 503        and optional roll, pitch, and yaw offsets, and sends the resulting target
 504        pose to MoveIt.
 505
 506        Args:
 507            request (RelativeMove.Request): Service request containing position
 508                and optional orientation deltas and movement parameters.
 509            response (RelativeMove.Response): Service response populated with the
 510                movement result.
 511
 512        Returns:
 513            RelativeMove.Response: The populated service response.
 514        """
 515        
 516        vx = request.vx
 517        vy = request.vy
 518        vz = request.vz
 519        self.get_logger().info(f"Service Call: Relative Move by Vector [{vx}, {vy}, {vz}]")
 520        self.current_state = ExtendedStatus.STATE_BUSY
 521        self.status_text = f"Executing relative move by vector [{vx}, {vy}, {vz}]..."
 522        self.publish_status()
 523
 524        try:
 525            # Look up current pose of the tool frame
 526            now = rclpy.time.Time()
 527            trans = self.tf_buffer.lookup_transform(BASE_FRAME, TOOL_FRAME, now, timeout=rclpy.duration.Duration(seconds=1.0))
 528
 529            curr_x = trans.transform.translation.x
 530            curr_y = trans.transform.translation.y
 531            curr_z = trans.transform.translation.z
 532
 533            target_x = curr_x + vx
 534            target_y = curr_y + vy
 535            target_z = curr_z + vz
 536
 537            self.get_logger().info(f"Calculated target: {target_x:.3f}, {target_y:.3f}, {target_z:.3f}")
 538
 539            has_orientation = request.has_orientation
 540            target_roll = target_pitch = target_yaw = 0.0
 541            if has_orientation:
 542                # quaternion_to_euler uses asin() for pitch, capped at +-90deg.
 543                # If the arm is already near that boundary (e.g. right after
 544                # tilted_for_pour), roll/yaw become coupled and this delta
 545                # composition gives unintuitive results. Fine for small nudges
 546                # from a normal pose, which is the expected use case.
 547                q = trans.transform.rotation
 548                curr_roll, curr_pitch, curr_yaw = quaternion_to_euler(q.x, q.y, q.z, q.w)
 549                target_roll = curr_roll + request.roll_delta
 550                target_pitch = curr_pitch + request.pitch_delta
 551                target_yaw = curr_yaw + request.yaw_delta
 552                self.get_logger().info(
 553                    f"Calculated target orientation (rpy): {target_roll}, {target_pitch}, {target_yaw}"
 554                )
 555
 556            if self.send_goal(
 557                target_x, target_y, target_z,
 558                has_orientation=has_orientation,
 559                roll=target_roll,
 560                pitch=target_pitch,
 561                yaw=target_yaw,
 562                motion_params=request.motion_params,
 563            ):
 564                self._await_action(
 565                    self.arm_movement_finished,
 566                    self.ACTION_TIMEOUT_SEC,
 567                    'arm_action_successful',
 568                    'arm_action_message',
 569                    f"Relative movement to ({target_x}, {target_y}, {target_z})",
 570                    response
 571                )
 572            else:
 573                response.success = False
 574                response.message = "Failed to initiate relative movement (action server unavailable)"
 575
 576        except Exception as e:
 577            self.get_logger().error(f"Could not calculate relative move: {e}")
 578            response.success = False
 579            response.message = f"Relative move TF lookup failed: {e}"
 580
 581        return self.finalize_service_status(response)

 620    # --- Orientation / Motion Params Helpers ---
 621    def clamp_motion_params(self, motion_params):
 622        """Clamp velocity and acceleration scaling factors to the valid range.
 623
 624        Values outside the MoveIt scaling range of 0.0 to 1.0 are clamped and
 625        a warning is logged.
 626
 627        Args:
 628            motion_params: Movement parameter object containing
 629                ``velocity_scale`` and ``acceleration_scale``.
 630
 631        Returns:
 632            tuple[float, float]: Clamped velocity and acceleration scaling
 633                factors.
 634        """
 635
 636        velocity_scale = motion_params.velocity_scale
 637        acceleration_scale = motion_params.acceleration_scale
 638
 639        if velocity_scale < 0.0 or velocity_scale > 1.0:
 640            self.get_logger().warn(f"velocity_scale {velocity_scale} out of range, clamping to [0.0, 1.0]")
 641            velocity_scale = max(0.0, min(1.0, velocity_scale))
 642
 643        if acceleration_scale < 0.0 or acceleration_scale > 1.0:
 644            self.get_logger().warn(f"acceleration_scale {acceleration_scale} out of range, clamping to [0.0, 1.0]")
 645            acceleration_scale = max(0.0, min(1.0, acceleration_scale))
 646
 647        return velocity_scale, acceleration_scale

 649    # --- Action Client Methods ---
 650    def send_goal(self, x, y, z, has_orientation=False, roll=0.0, pitch=0.0, yaw=0.0, motion_params=None):
 651        """Send a Cartesian movement goal to MoveIt.
 652
 653        Constructs a position constraint for the robot tool and optionally adds
 654        an orientation constraint and motion-scaling parameters. The goal is
 655        submitted asynchronously.
 656
 657        This method reports whether the goal could be initiated; it does not
 658        indicate whether the physical movement has completed successfully.
 659        Completion is reported through the action callbacks.
 660
 661        Args:
 662            x (float): Target X position in ``BASE_FRAME``.
 663            y (float): Target Y position in ``BASE_FRAME``.
 664            z (float): Target Z position in ``BASE_FRAME``.
 665            has_orientation (bool): Whether an orientation constraint should be
 666                included.
 667            roll (float): Target roll angle in radians.
 668            pitch (float): Target pitch angle in radians.
 669            yaw (float): Target yaw angle in radians.
 670            motion_params: Optional movement parameters containing velocity and
 671                acceleration scaling factors.
 672
 673        Returns:
 674            bool: ``True`` if the MoveIt action server is available and the goal
 675                was submitted; ``False`` if the action server is unavailable.
 676        """
 677
 678        if not self.arm_client.wait_for_server(timeout_sec=self.SERVER_WAIT_TIMEOUT_SEC):
 679            self.get_logger().error('Arm server not available')
 680            return False
 681
 682        goal_msg = MoveGroup.Goal()
 683        goal_msg.request.group_name = self.PLANNING_GROUP
 684        goal_msg.request.num_planning_attempts = self.NUM_PLANNING_ATTEMPTS
 685        goal_msg.request.allowed_planning_time = self.ALLOWED_PLANNING_TIME_SEC
 686
 687        pos_constraint = PositionConstraint()
 688        pos_constraint.header.frame_id = BASE_FRAME
 689        pos_constraint.link_name = TOOL_FRAME
 690
 691        sphere = SolidPrimitive()
 692        sphere.type = SolidPrimitive.SPHERE
 693        sphere.dimensions = [self.SPHERE_TOLERANCE_RADIUS]
 694
 695        target_pose = Pose()
 696        target_pose.position.x = float(x)
 697        target_pose.position.y = float(y)
 698        target_pose.position.z = float(z)
 699
 700        pos_constraint.constraint_region.primitives.append(sphere)
 701        pos_constraint.constraint_region.primitive_poses.append(target_pose)
 702        pos_constraint.weight = 1.0
 703
 704        goal_constraints = Constraints()
 705        goal_constraints.position_constraints.append(pos_constraint)
 706
 707        if has_orientation:
 708            qx, qy, qz, qw = euler_to_quaternion(roll, pitch, yaw)
 709            orient_constraint = OrientationConstraint()
 710            orient_constraint.header.frame_id = BASE_FRAME
 711            orient_constraint.link_name = TOOL_FRAME
 712            orient_constraint.orientation.x = qx
 713            orient_constraint.orientation.y = qy
 714            orient_constraint.orientation.z = qz
 715            orient_constraint.orientation.w = qw
 716            orient_constraint.absolute_x_axis_tolerance = 0.1
 717            orient_constraint.absolute_y_axis_tolerance = 0.1
 718            orient_constraint.absolute_z_axis_tolerance = 0.1
 719            orient_constraint.weight = 1.0
 720            goal_constraints.orientation_constraints.append(orient_constraint)
 721
 722        goal_msg.request.goal_constraints.append(goal_constraints)
 723
 724        if motion_params is not None:
 725            velocity_scale, acceleration_scale = self.clamp_motion_params(motion_params)
 726            if velocity_scale > 0.0:
 727                goal_msg.request.max_velocity_scaling_factor = velocity_scale
 728            if acceleration_scale > 0.0:
 729                goal_msg.request.max_acceleration_scaling_factor = acceleration_scale
 730
 731        self.arm_movement_finished.clear()
 732        future = self.arm_client.send_goal_async(
 733            goal_msg,
 734            feedback_callback=self.arm_feedback_callback
 735        )
 736        future.add_done_callback(self.goal_response_callback)
 737        return True

 893        Args:
 894            future: Future containing the completed MoveIt action result.
 895
 896        Returns:
 897            None
 898        """
 899
 900        try:
 901            result = future.result().result
 902            error_code = result.error_code.val
 903
 904            if error_code == result.error_code.SUCCESS:
 905                self.get_logger().info('Movement complete!')
 906                self.arm_action_successful = True
 907                self.arm_action_message = 'Movement complete'
 908            else:
 909                self.arm_action_successful = False
 910
 911                match error_code:
 912                    case result.error_code.NO_IK_SOLUTION:
 913                        msg = "Coordinates out of reach (no inverse kinematics solution)"
 914                    case result.error_code.PLANNING_FAILED:
 915                        msg = "Planning failed (path blocked by obstacle or self-collision)"
 916                    case result.error_code.TIMED_OUT:
 917                        msg = "MoveIt planning/movement timed out"
 918                    case result.error_code.GOAL_IN_COLLISION:
 919                        msg = "Goal is in collision (target position inside an obstacle)"
 920                    case result.error_code.START_STATE_IN_COLLISION:
 921                        msg = "Start state is in collision (robot currently in collision)"
 922                    case result.error_code.CONTROL_FAILED:
 923                        msg = "Control failed during execution (hardware error)"
 924                    case result.error_code.ABORT:
 925                        msg = "Movement was aborted by MoveIt"
 926                    case _:
 927                        msg = f"MoveIt failed with error code: {error_code}"
 928
 929                self.get_logger().error(f"ERROR: {msg}")
 930                self.arm_action_message = msg
 931                self.handle_moveit_failure()
